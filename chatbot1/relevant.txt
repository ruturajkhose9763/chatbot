def _build_dashboard_context():
    # Dashboard data must be defensive: an older/stale session or an older
    # JSON record should never turn /admin or /teacher into a 500 page.
    content = get_content() or {}
    raw_admissions = (get_admissions() or {}).get("submissions", [])
    admissions = sorted(
        [a for a in raw_admissions if isinstance(a, dict)],
        key=lambda x: x.get("date", ""),
        reverse=True,
    )
    accounts = (get_accounts() or {}).get("accounts", [])
    college = get_data() or {}
    current_account = find_account(session.get("username", "")) or {
        "username": session.get("username", ""),
        "full_name": session.get("full_name", "Admin"),
        "role": session.get("role", "main_admin"),
        "class_key": None,
    }
    classrooms = (get_classrooms() or {}).get("classes", {})
    if not isinstance(classrooms, dict):
        classrooms = {}

    my_class = None
    if current_account.get("role") == "staff" and current_account.get("class_key"):
        my_class = classrooms.get(current_account["class_key"])

    subjects_data = get_subjects() or {"branches": {}, "study_material_link": ""}
    attendance_data = get_attendance() or {"classes": {}}
    my_class_students = []
    my_class_student_records = []
    my_class_attendance_dates = []
    my_class_attendance_pct = []
    my_class_attendance_table = []
    if my_class is not None:
        class_key = current_account.get("class_key")
        class_att = attendance_data.get("classes", {}).get(class_key, {"students": [], "records": {}})
        if not isinstance(class_att, dict):
            class_att = {"students": [], "records": {}}
        my_class_students = class_att.get("students", [])
        profiles = [s for s in get_students().get("students", []) if s.get("class_key") == class_key]
        profile_by_name = {str(s.get("name", "")).strip().lower(): s for s in profiles}
        for raw in my_class_students:
            if isinstance(raw, dict):
                my_class_student_records.append(raw)
            else:
                p = profile_by_name.get(str(raw).strip().lower())
                my_class_student_records.append(p or {"name": str(raw), "roll_no": "", "mobile": ""})
        records = class_att.get("records", {})
        my_class_attendance_dates = sorted(records.keys(), reverse=True)[:10]

        # Graph data: % of recorded days each student was present.
        for student in my_class_students:
            student_name = student.get("name", "") if isinstance(student, dict) else str(student)
            total_days = 0
            present_days = 0
            for day_status in records.values():
                if student_name in day_status:
                    total_days += 1
                    if day_status[student_name] == "present":
                        present_days += 1
            pct = round((present_days / total_days) * 100) if total_days else 0
            my_class_attendance_pct.append({"name": student_name, "pct": pct, "present_days": present_days, "total_days": total_days})

        for student in my_class_students:
            student_name = student.get("name", "") if isinstance(student, dict) else str(student)
            profile = profile_by_name.get(student_name.strip().lower(), {})
            row = {"name": student_name, "roll_no": profile.get("roll_no", ""), "statuses": []}
            for d in my_class_attendance_dates:
                row["statuses"].append(records.get(d, {}).get(student_name, "-"))
            my_class_attendance_table.append(row)

    is_main_admin = session.get("role") == "main_admin"

    analytics = None
    custom_qa_items = []
    if is_main_admin:
        raw_analytics = get_analytics() or {}
        analytics = {
            "total_questions": raw_analytics.get("total_questions", 0),
            "top_topics": sorted(raw_analytics.get("topic_counts", {}).items(), key=lambda x: x[1], reverse=True)[:8],
            "recent_questions": raw_analytics.get("recent_questions", [])[:15],
            "unanswered_questions": [
                q for q in raw_analytics.get("recent_questions", []) if q.get("topic") == "unmatched"
            ][:15],
        }
        custom_qa_items = get_custom_qa().get("items", [])

    # Teachers get a lighter version of the same AI Assistant info: just the
    # questions students asked that the chatbot couldn't answer, so they can
    # add a learned answer for their subject -- without full site analytics.
    teacher_unanswered = []
    if not is_main_admin:
        raw_analytics = get_analytics() or {}
        teacher_unanswered = [
            q for q in raw_analytics.get("recent_questions", []) if q.get("topic") == "unmatched"
        ][:15]
        custom_qa_items = get_custom_qa().get("items", [])

    # ---- Read-only dashboard summary stats (derived from existing data, no schema changes) ----
    dash_stats = None
    if is_main_admin:
        try:
            att_all = get_attendance().get("classes", {})
            total_students = sum(len(c.get("students", [])) for c in att_all.values())
        except Exception:
            total_students = 0
        total_departments = len({c.get("branch") for c in classrooms.values() if c.get("branch")})
        total_courses = len(college.get("courses_offered", []))
        dash_stats = {
            "total_students": total_students,
            "total_staff": len(accounts),
            "total_departments": total_departments,
            "total_courses": total_courses,
            "chatbot_queries": analytics["total_questions"] if analytics else 0,
            "total_admissions": len(admissions),
        }

    study_materials = (get_study_materials() or {}).get("items", [])
    if not is_main_admin:
        study_materials = [m for m in study_materials if m.get("uploaded_by") == session["username"]]

    return dict(
        content=content, admissions=admissions, accounts=accounts,
        college=college, current_account=current_account,
        is_main_admin=is_main_admin,
        my_class=my_class, my_class_key=current_account.get("class_key"),
        my_class_students=my_class_students, my_class_student_records=my_class_student_records, my_class_attendance_dates=my_class_attendance_dates,
        my_class_attendance_pct=my_class_attendance_pct, my_class_attendance_table=my_class_attendance_table,
        all_classes=classrooms, analytics=analytics, custom_qa_items=custom_qa_items,
        teacher_unanswered=teacher_unanswered,
        subjects_data=subjects_data, dash_stats=dash_stats, study_materials=study_materials
    )

@app.route("/admin/add-custom-qa", methods=["POST"])
@login_required
def add_custom_qa():
    question = (request.form.get("question") or "").strip()
    answer = (request.form.get("answer") or "").strip()
    if not question or not answer:
        flash("Please provide both the question and an answer.")
        return redirect_to_dashboard()

    qa_data = get_custom_qa()
    items = qa_data.get("items", [])
    new_id = (max((item.get("id", 0) for item in items), default=0) + 1)
    items.append({
        "id": new_id,
        "question": question,
        "answer": answer,
        "keywords": extract_keywords(question),
        "added_on": datetime.now().strftime("%d %b %Y, %I:%M %p"),
    })
    qa_data["items"] = items
    save_custom_qa(qa_data)
    flash("Chatbot taught this answer — it will now recognise similar questions.")
    return redirect_to_dashboard()

@app.route("/admin/upload-dp", methods=["POST"])
@login_required
def upload_dp():
    file = request.files.get("dp")
    if not file or file.filename == "" or not allowed_image(file.filename):
        flash("Please choose a valid image file (PNG/JPG/JPEG/WEBP).")
        return redirect_to_dashboard()

    try:
        result = cloudinary.uploader.upload(file, folder="college_dp")
        dp_url = result["secure_url"]
    except Exception as e:
        flash(f"Photo upload failed: {e}")
        return redirect_to_dashboard()

    accounts_data = get_accounts()
    for acc in accounts_data["accounts"]:
        if acc["username"] == session["username"]:
            acc["dp_filename"] = dp_url  # now stores a full Cloudinary URL
    save_json(STAFF_ACCOUNTS_PATH, accounts_data)

    flash("Profile photo updated.")
    return redirect_to_dashboard()

@app.route("/admin/attendance/add-student", methods=["POST"])
@class_teacher_required
def attendance_add_student():
    account = find_account(session["username"])
    class_key = account.get("class_key")
    student_name = (request.form.get("student_name") or "").strip()
    roll_no = normalize_roll(request.form.get("roll_no"))
    mobile = (request.form.get("mobile") or "").strip()
    pin = (request.form.get("pin") or "").strip()
    if not student_name or not roll_no or not mobile or len(pin) < 6:
        flash("Student name, roll number, mobile number and a PIN of at least 6 characters are required.")
        return redirect_to_dashboard()
    existing = find_student(roll_no)
    if existing and existing.get("class_key") != class_key:
        flash("That roll number is already assigned to another class.")
        return redirect_to_dashboard()
    students = get_students()
    if existing:
        mobile_changed = existing.get("mobile", "") != mobile
        existing.update({"name": student_name, "mobile": mobile, "class_key": class_key})
        if mobile_changed:
            existing["mobile_verified"] = False
        if pin:
            existing["pin_hash"] = generate_password_hash(pin)
    else:
        students["students"].append({
            "id": secrets.token_hex(8), "roll_no": roll_no, "name": student_name,
            "mobile": mobile, "class_key": class_key, "pin_hash": generate_password_hash(pin),
            "active": True, "mobile_verified": False, "otp_hash": None, "otp_expires": None,
            "notifications": [], "created_at": datetime.now().strftime("%Y-%m-%d %H:%M")
        })
    # Persist the new/updated profile before looking it up again. The previous
    # version tried find_student() before saving a newly-created profile, so a
    # new student's OTP flow was silently skipped.
    save_students(students)
    target = find_student(roll_no)
    if target and target.get("mobile") and not target.get("mobile_verified", False):
        otp = f"{secrets.randbelow(1000000):06d}"
        target["otp_hash"] = generate_password_hash(otp)
        target["otp_expires"] = (datetime.now() + timedelta(minutes=10)).timestamp()
        save_students(students)
        if _send_teacher_otp_sms(target["mobile"], otp):
            flash("Student account created. Verification OTP sent to the registered mobile number.")
        else:
            flash("Student account created. Configure Twilio for real SMS OTP delivery; the account remains pending verification.")
            print(f"Student OTP for {roll_no}: {otp}")
    else:
        # Profile was already saved above.
        pass
    attendance = get_attendance()
    cls = attendance["classes"].setdefault(class_key, {"students": [], "records": {}})
    if student_name not in cls["students"]:
        cls["students"].append(student_name)
        save_attendance(attendance)
    flash(f"Student {student_name} (Roll No. {roll_no}) is ready for Student Portal login.")
    return redirect_to_dashboard()

@app.route("/admin/attendance/mark", methods=["POST"])
@class_teacher_required
def attendance_mark():
    account = find_account(session["username"])
    class_key = account.get("class_key")
    att_date = request.form.get("date") or datetime.now().strftime("%Y-%m-%d")
    attendance = get_attendance()
    cls = attendance["classes"].setdefault(class_key, {"students": [], "records": {}})
    students = get_students()
    teacher_name = account.get("full_name", "Teacher")
    day_record = {}
    for raw_student in cls.get("students", []):
        student_name = raw_student.get("name", "") if isinstance(raw_student, dict) else str(raw_student)
        status = request.form.get(f"status_{student_name}", "absent")
        day_record[student_name] = status
        profile = find_student_by_name_class(student_name, class_key)
        if profile:
            profile.setdefault("notifications", []).insert(0, _new_notification(
                "Attendance marked",
                f"Your attendance for {att_date} has been marked {status.title()} by {teacher_name}.",
                "attendance"
            ))
            profile["notifications"] = profile["notifications"][:50]
    cls["records"][att_date] = day_record
    save_attendance(attendance)
    save_students(students)
    flash(f"Attendance saved for {att_date}. Students can see the update in their Student Portal.")
    return redirect_to_dashboard()


@app.route("/student/login", methods=["GET", "POST"])
def student_login():
    if request.method == "POST":
        roll_no = normalize_roll(request.form.get("roll_no"))
        pin = request.form.get("pin", "")
        student = find_student(roll_no)
        if student and student.get("active", True) and student.get("pin_hash") and check_password_hash(student["pin_hash"], pin):
            if student.get("mobile") and not student.get("mobile_verified", False):
                flash("Student mobile verification is pending. Verify the registered mobile number first.")
                return redirect(url_for("student_verify", roll_no=roll_no))
            session["student_logged_in"] = True
            session["student_roll_no"] = roll_no
            return redirect(url_for("student_dashboard"))
        flash("Invalid Roll Number or PIN.")
    return render_template("student_login.html", college=get_data())


@app.route("/student")
def student_dashboard():
    if not session.get("student_logged_in"):
        return redirect(url_for("student_login"))
    student = find_student(session.get("student_roll_no"))
    if not student:
        session.pop("student_logged_in", None)
        session.pop("student_roll_no", None)
        return redirect(url_for("student_login"))
    attendance = get_attendance().get("classes", {}).get(student.get("class_key"), {"students": [], "records": {}})
    records = attendance.get("records", {})
    name = student.get("name", "")
    dates=[]; present=0; total=0
    history=[]
    for date_key in sorted(records.keys(), reverse=True):
        status = records[date_key].get(name)
        if status in ("present", "absent"):
            total += 1
            present += status == "present"
            history.append({"date": date_key, "status": status})
    pct = round((present / total) * 100, 1) if total else 0
    unread = sum(1 for n in student.get("notifications", []) if not n.get("read"))
    return render_template("student.html", college=get_data(), student=student,
                           attendance_pct=pct, present_days=present, total_days=total,
                           history=history[:30], notifications=student.get("notifications", [])[:30], unread=unread)

@app.route("/student/verify", methods=["GET", "POST"])
def student_verify():
    roll_no = normalize_roll(request.args.get("roll_no") or request.form.get("roll_no"))
    student = find_student(roll_no)
    if not student:
        flash("Student account not found.")
        return redirect(url_for("student_login"))
    if student.get("mobile_verified"):
        return redirect(url_for("student_login"))
    if request.method == "POST":
        otp = (request.form.get("otp") or "").strip()
        expires = student.get("otp_expires", 0) or 0
        if otp and student.get("otp_hash") and datetime.now().timestamp() < expires and check_password_hash(student["otp_hash"], otp):
            data = get_students()
            target = find_student(roll_no)
            target["mobile_verified"] = True
            target["otp_hash"] = None
            target["otp_expires"] = None
            save_students(data)
            flash("Student mobile verified. You can now log in.")
            return redirect(url_for("student_login"))
        flash("Invalid or expired OTP.")
    return render_template("student_verify.html", college=get_data(), roll_no=roll_no, mobile=student.get("mobile", ""))

@app.route("/admin/subjects/set-teacher", methods=["POST"])
@main_admin_required
def subjects_set_teacher():
    branch = request.form.get("branch", "")
    semester = request.form.get("semester", "")
    subject = request.form.get("subject", "")
    teacher_name = (request.form.get("teacher_name") or "").strip()

    subjects_data = get_subjects()
    branch_data = subjects_data.get("branches", {}).get(branch)
    if not branch_data or semester not in branch_data.get("semesters", {}):
        flash("Invalid branch or semester.")
        return redirect_to_dashboard()

    branch_data["semesters"][semester].setdefault("teachers", {})
